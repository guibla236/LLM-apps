import json
from datasets import Dataset

SEED = 42

# ASUMIMOS QUE SUBISTE A COLAB UN ARCHIVO tickets.json CON LA COMBINACIÓN DE TUS TICKETS
try:
    with open('/kaggle/input/datasets/guibla236/tickets-json/tickets.json', 'r', encoding='utf-8') as f:
        tickets_data = json.load(f)
except FileNotFoundError:
    print("Archivo tickets.json no encontrado en Colab (por favor sube tus datos reales).")
    raise FileNotFoundError("Provide tickets.json")

# Q&A Pairs generation for the tickets (Synthetic Augmentation)
qa_pairs = []
for t in tickets_data:
    desc = (t.get("description", "") or "").strip()
    impact = (t.get("impact", "") or "").strip()
    actions = (t.get("actions", "") or "").strip()

    # Evita entrenar con ejemplos vacíos o de poco valor
    if not desc or not actions:
        continue

    # Template 1
    qa_pairs.append({
        "question": f"Tengo un problema: {desc}. ¿Qué pasos técnicos debo seguir para solucionarlo?",
        "ground_truth": f"Las acciones recomendadas para resolver este incidente son: {actions}. El impacto asociado es: {impact if impact else 'No especificado'}."
    })

    # Template 2
    qa_pairs.append({
        "question": f"Si ocurre la siguiente situación: {desc}. ¿Cuál es el procedimiento técnico y qué riesgo implica?",
        "ground_truth": f"Procedimiento recomendado: {actions}. Riesgo o impacto asociado: {impact if impact else 'No especificado'}."
    })

    # Template 3
    qa_pairs.append({
        "question": f"Resume las acciones de soporte y el impacto relacionado con este reporte: {desc}",
        "ground_truth": f"Acciones: {actions}. Impacto: {impact if impact else 'No especificado'}."
    })

dataset = Dataset.from_list(qa_pairs)

def format_instruction(sample):
    sys_prompt = (
        "Rol: Eres un asistente experto en soporte técnico IT. "
        "Analiza el problema reportado por el usuario y responde de forma concisa indicando: "
        "1) acciones técnicas recomendadas para resolverlo y "
        "2) riesgos o impacto asociados."
    )
    return (
        f"<|im_start|>system\n{sys_prompt}<|im_end|>\n"
        f"<|im_start|>user\n{sample['question']}<|im_end|>\n"
        f"<|im_start|>assistant\n{sample['ground_truth']}<|im_end|>"
    )

def encode_data(sample):
    return tokenizer(
        format_instruction(sample),
        truncation=True,
        padding="max_length",
        max_length=512
    )

tokenized_dataset = dataset.map(encode_data, batched=False)

# Split train/validation para monitorear generalización
split_dataset = tokenized_dataset.train_test_split(test_size=0.1, seed=SEED)
train_dataset = split_dataset["train"]
eval_dataset = split_dataset["test"]

print(f"Total ejemplos: {len(tokenized_dataset)}")
print(f"Train: {len(train_dataset)} | Eval: {len(eval_dataset)}")
